Validação de Formulários

Validação HTML5 Nativa
O que é: Validação incorporada no HTML5 usando atributos como required, pattern, min/max, type, etc.
Características: Implementação direta sem JavaScript, porém com opções de personalização limitadas.
Preencha o formulário...
Código HTML5
<form id="html5Form">
    <!-- Email obrigatório -->
    <input type="email" required placeholder="seu@email.com">

    <!-- Telefone com padrão regex -->
    <input type="tel" pattern="^\(\d{2}\)\s\d{4,5}-\d{4}$"
           placeholder="(11) 99999-9999">

    <!-- Idade com limites -->
    <input type="number" min="18" max="100" required>

    <button type="submit">Validar</button>
</form>
Validação em Tempo Real
O que é: Validação que acontece enquanto o usuário digita, usando eventos como 'input' e 'blur'.
Características: Oferece feedback durante a digitação, requerendo mais código JavaScript.
Digite uma senha...
  • Mínimo 8 caracteres
  • Uma letra maiúscula
  • Uma letra minúscula
  • Um número
  • Um caractere especial
Digite nos campos acima...
Código JavaScript (Tempo Real)
// Validação de email em tempo real
document.getElementById('realtimeEmail').addEventListener('input', function() {
    const email = this.value;
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

    if (email.length === 0) {
        resetField(this);
    } else if (emailRegex.test(email)) {
        setValid(this, 'Email válido!');
    } else {
        setInvalid(this, 'Email inválido');
    }
});

// Validação de força de senha
document.getElementById('realtimePassword').addEventListener('input', function() {
    const password = this.value;
    checkPasswordStrength(password);
});

function checkPasswordStrength(password) {
    const requirements = [
        { id: 'req-length', test: password.length >= 8 },
        { id: 'req-upper', test: /[A-Z]/.test(password) },
        { id: 'req-lower', test: /[a-z]/.test(password) },
        { id: 'req-number', test: /\d/.test(password) },
        { id: 'req-special', test: /[!@#$%^&*(),.?":{}|<>]/.test(password) }
    ];

    let validCount = 0;
    requirements.forEach(req => {
        const element = document.getElementById(req.id);
        if (req.test) {
            element.classList.add('valid');
            validCount++;
        } else {
            element.classList.remove('valid');
        }
    });

    updateStrengthBar(validCount);
}

Diferenças Principais
// HTML5 - Automática pelo navegador
<input type="email" required>

// JavaScript - Controle manual
input.addEventListener('input', function() {
    if (isValid(this.value)) {
        showSuccess();
    } else {
        showError();
    }
});
